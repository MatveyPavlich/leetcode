```
import pandas as pd

def movie_rating(movies: pd.DataFrame, users: pd.DataFrame, movie_rating: pd.DataFrame) -> pd.DataFrame:
    df = (
        movie_rating[["user_id"]]
        .groupby("user_id", as_index=False)
        .agg(
            n_rating=("user_id", "count")
        )
        .merge(
            users,
            on="user_id",
            how="left"
        )
        .sort_values(by=["n_rating", "name"], ascending=[False, True])
        .iloc[:1]
        .drop(columns=["user_id", "n_rating"])
    )

    movie = (
        movie_rating[movie_rating["created_at"].dt.to_period("M") == "2020-02"]
        .groupby("movie_id", as_index=False)
        .agg(
            avg_rating=("rating", "mean")
        )
        .merge(
            movies,
            on="movie_id",
            how="left"
        )
        .sort_values(by=["avg_rating", "title"], ascending=[False, True])
        .drop(columns=["movie_id", "avg_rating"])
        .iloc[:1]
    )

    df = df.rename(columns={"name": "results"})
    movie = movie.rename(columns={"title": "results"})

    return pd.concat([df, movie], ignore_index=True)
```
