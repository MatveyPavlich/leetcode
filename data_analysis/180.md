```
import pandas as pd

def consecutive_numbers(logs: pd.DataFrame) -> pd.DataFrame:
    logs["next_id"] = logs["id"] + 1
    logs["nnext_id"] = logs["id"] + 2
    df_1 = logs[["id", "num"]].merge(
        logs,
        left_on="id",
        right_on="next_id",
        how="left",
        suffixes=("", "_next")
    )
    df_2 = df_1[["id", "num", "next_id", "num_next"]].merge(
        logs,
        left_on="id",
        right_on="nnext_id",
        how="left",
        suffixes=("", "_nnext")
    )
    return pd.DataFrame({
        "ConsecutiveNums" : (
            df_2["num"][
                (df_2["num"] == df_2["num_next"]) & (df_2["num"] == df_2["num_nnext"])
            ].drop_duplicates()
        )
    })
```
