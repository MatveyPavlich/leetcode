```
import pandas as pd

def project_employees_i(project: pd.DataFrame, employee: pd.DataFrame) -> pd.DataFrame:
    df = pd.merge(project, employee, how="left", on="employee_id")
    df_final = (
        df
        .groupby("project_id")
        .agg(
            average_years=("experience_years", "mean")
        )
        .reset_index()
    )
    df_final["average_years"] = df_final["average_years"].round(2)
    return df_final[["project_id", "average_years"]]
```
