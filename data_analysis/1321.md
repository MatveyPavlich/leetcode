# Pandas
```
import pandas as pd

def restaurant_growth(customer: pd.DataFrame) -> pd.DataFrame:
    agg = (
        customer
        .groupby("visited_on")
        .agg({"amount": "sum"})
        .sort_index()
        .reset_index()
    )
    agg["average_amount"] = agg["amount"].rolling(7).mean().round(2)
    agg["amount"] = agg["amount"].rolling(7).sum()
    return agg[agg["amount"].notna()][["visited_on", "amount", "average_amount"]]
```
