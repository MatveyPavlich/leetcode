```
import pandas as pd

def combine_two_tables(person: pd.DataFrame, address: pd.DataFrame) -> pd.DataFrame:
    out = (
        pd.merge(
            person,
            address,
            on="personId",
            how="left"
        )
        .drop(columns=["personId", "addressId"])
    )

    return out[["firstName", "lastName", "city", "state"]]
```
