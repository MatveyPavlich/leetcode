## First try
```
import pandas as pd

def biggest_single_number(my_numbers: pd.DataFrame) -> pd.DataFrame:
    my_numbers["ttl"] = my_numbers.groupby("num")["num"].transform("count")
    my_numbers = my_numbers[my_numbers["ttl"] == 1]
    if len(my_numbers) > 0:
        my_numbers["biggest"] = my_numbers["num"].max()
        my_numbers = my_numbers[my_numbers["num"] == my_numbers["biggest"]]
    else:
        my_numbers = pd.DataFrame({
            "num": [None]
        })
    return my_numbers[["num"]]
```

## Try 2:
```
import pandas as pd

def biggest_single_number(my_numbers: pd.DataFrame) -> pd.DataFrame:
    my_numbers["ttl"] = my_numbers.groupby("num")["num"].transform("count")
    my_numbers = my_numbers[my_numbers["ttl"] == 1]
    if my_numbers.empty:
        return pd.DataFrame({"num": [None]})
    return pd.DataFrame({"num": [my_numbers["num"].max()]})
```

## One of the solutions
```
import pandas as pd

def biggest_single_number(my_numbers: pd.DataFrame) -> pd.DataFrame:
    return my_numbers.drop_duplicates(keep = False).max().to_frame(name = 'num')
```
