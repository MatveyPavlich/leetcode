```
import pandas as pd

def delete_duplicate_emails(person: pd.DataFrame) -> None:
    person.sort_values(by=["email", "id"], inplace=True)
    person["row_num"] = person.groupby("email").cumcount()
    person.drop(person[person["row_num"] > 0].index, inplace=True)
    person.drop(columns="row_num", inplace=True)
```
