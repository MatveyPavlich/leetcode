import pandas as pd

def find_investments(insurance: pd.DataFrame) -> pd.DataFrame:
    # Apply .duplacated() twice
    # For ['lon', 'lat'] pair we inverse it with '~' as we need to remove duplicates
    # For 'tiv_2015' we apply it as is
    result = insurance[(insurance.duplicated('tiv_2015', keep=False)) & (~insurance.duplicated(['lat', 'lon'], keep=False))]

    return pd.DataFrame({'tiv_2016': [round(result['tiv_2016'].sum(), 2)]})
