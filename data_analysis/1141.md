```
import pandas as pd

def user_activity(activity: pd.DataFrame) -> pd.DataFrame:
    ref = pd.Timestamp("2019-07-27")

    activity = activity[
        (activity["activity_date"] >= ref - pd.Timedelta(days=29)) &
        (activity["activity_date"] <= ref)
    ]
    out = (
        activity
        .groupby("activity_date")
        .agg(
            active_users=("user_id", "nunique")
        )
        .reset_index()
        .rename(columns={"activity_date": "day"})
    )

    return out[["day", "active_users"]]
```
