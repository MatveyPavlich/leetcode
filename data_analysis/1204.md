# Pandas
```
import pandas as pd

def last_passenger(queue: pd.DataFrame) -> pd.DataFrame:
    queue.sort_values(by="turn", ascending=True, inplace=True)
    queue["cum_weight"] = queue["weight"].cumsum()
    queue = queue[queue["cum_weight"] <= 1000]
    return queue.iloc[[-1]][["person_name"]]
```

```
import pandas as pd

def last_passenger(queue: pd.DataFrame) -> pd.DataFrame:
    queue.sort_values('turn', inplace=True)
    return queue.loc[queue.weight.cumsum() <= 1000, ['person_name']].tail(1)
```
