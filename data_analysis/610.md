import pandas as pd

def triangle_judgement(triangle: pd.DataFrame) -> pd.DataFrame:

  triangle['triangle'] = triangle.apply(
                       lambda x: ['No','Yes'][sum(x) > 2 * max(x)] , axis = 1)
  return triangle
