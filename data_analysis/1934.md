import pandas as pd

def confirmation_rate(signups: pd.DataFrame, confirmations: pd.DataFrame) -> pd.DataFrame:
    df = pd.merge(signups, confirmations, how="left", on="user_id", suffixes=("", "_con"))
    df["conf_msg"] = df["time_stamp_con"].where(df["action"] == "confirmed")
    df_agg = (
        df
        .groupby("user_id")
        .agg(
            ttl_msg=("time_stamp_con", "count"),
            ttl_con=("conf_msg", "count")
        )
        .reset_index()
    )
    df_agg["confirmation_rate"] = (
        df_agg["ttl_con"]
        .div(df_agg["ttl_msg"])
        .fillna(0)
        .round(2)
    )
    return df_agg[["user_id", "confirmation_rate"]]
