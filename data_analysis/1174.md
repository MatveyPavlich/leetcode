import pandas as pd

def immediate_food_delivery(delivery: pd.DataFrame) -> pd.DataFrame:
    delivery["first_order"] = delivery.groupby("customer_id")["order_date"].transform("min")
    delivery = delivery[delivery["first_order"] == delivery["order_date"]]
    delivery["_is_immediate"] = (delivery["order_date"] == delivery["customer_pref_delivery_date"]).astype(int)
    ttl_first_orders = len(delivery)
    df = (
        delivery
        .agg(
            immediate_percentage=("_is_immediate", "sum")
        )
        .reset_index()
    )
    df["immediate_percentage"] = (
        (df["_is_immediate"] * 100) / ttl_first_orders
    ).round(2)

    return df[["immediate_percentage"]]
