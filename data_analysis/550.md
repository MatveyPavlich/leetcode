```
import pandas as pd

def gameplay_analysis(activity: pd.DataFrame) -> pd.DataFrame:
    activity["first_day"] = activity.groupby("player_id")["event_date"].transform("min")
    activity["_2nd_day"] = (
        (activity["event_date"] - activity["first_day"]).dt.days == 1
    ).astype(int)
    
    return pd.DataFrame({
        "fraction": [(activity["_2nd_day"].sum() / activity["player_id"].nunique()).round(2)]
    })
```
