# Pandas
```
import pandas as pd

def find_customers(customer: pd.DataFrame, product: pd.DataFrame) -> pd.DataFrame:
    n_product = product["product_key"].nunique()
    df_agg = (
        customer
        .groupby("customer_id")
        .agg({"product_key": "nunique"})
        .query("product_key == @n_product")
        .reset_index()
        .drop(columns=["product_key"])
    )
    return df_agg
```
