# Pandas
```
import pandas as pd

def count_employees(employees: pd.DataFrame) -> pd.DataFrame:
    df = (
        pd.merge(
            employees[["employee_id", "name"]],
            employees[["reports_to", "age"]],
            left_on="employee_id",
            right_on="reports_to",
            how="inner"
        )
        .groupby(["employee_id", "name"], as_index=False)
        .agg(
            reports_count=("employee_id", "count"),
            average_age=("age", "mean")
        )
    )
    df["average_age"] = df["average_age"].add(1e-9).round()
    
    return df
```
