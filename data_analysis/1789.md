```
import pandas as pd

def find_primary_department(employee: pd.DataFrame) -> pd.DataFrame:
    employee["dep_count"] = employee.groupby("employee_id")["department_id"].transform("count")
    return (
        employee[
            (employee["dep_count"] == 1) |
            (employee["primary_flag"] == 'Y')
        ][["employee_id", "department_id"]]
    )
```
