# Pandas
```
import pandas as pd

def categorize_products(activities: pd.DataFrame) -> pd.DataFrame:
    return (
        activities
        .drop_duplicates()
        .groupby(["sell_date"], as_index=False)
        .agg(
            num_sold=("product", "count"),
            products=("product", lambda x: ",".join(sorted(x)))
        )
    )
```
