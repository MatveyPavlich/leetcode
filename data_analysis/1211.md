```
def queries_stats(queries: pd.DataFrame) -> pd.DataFrame:
    df = queries.copy()
    df["ratio"] = df["rating"] / df["position"]
    df["lt3"] = df["rating"] < 3
    # df["poor_queries"] = (
    #     100 * (df["rating"] < 3).sum() / df["query_name"].count()
    # ).round(2)
    df_agg = (
        df
        .groupby("query_name")
        .agg(
            quality=("ratio", "mean"),
            n_lt3=("lt3", "sum"),
            queries=("query_name", "count")
        )
        .reset_index()
    )
    # df_agg["quality"] = df_agg["quality"].round(2)
    df_agg["quality"] = df_agg["quality"].add(1e-8).round(2)
    df_agg["poor_query_percentage"] = (
        (100 * df_agg["n_lt3"]) / df_agg["queries"]
    ).round(2)

    return df_agg[["query_name", "quality", "poor_query_percentage"]]
```
