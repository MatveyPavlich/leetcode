```
import pandas as pd

def rising_temperature(weather: pd.DataFrame) -> pd.DataFrame:
    weather["prev_date"] = weather["recordDate"] - pd.Timedelta(days=1)

    df = weather.merge(
        weather,
        left_on="prev_date",
        right_on="recordDate",
        suffixes=("", "_prev")
    )

    df = df[df["temperature"] > df["temperature_prev"]]

    return df[["id"]].rename(columns={"id": "Id"})
```
