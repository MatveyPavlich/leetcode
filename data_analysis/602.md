# Pandas
```
import pandas as pd

def most_friends(request_accepted: pd.DataFrame) -> pd.DataFrame:
    ids = pd.concat([
        request_accepted["requester_id"],
        request_accepted["accepter_id"]
    ]).to_frame("id")

    return (
        ids
        .groupby("id", as_index=False)
        .agg(num=("id", "count"))
        .sort_values(by="num", ascending=False)
        .head(1)
    )
```
