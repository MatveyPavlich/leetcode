```
import pandas as pd

def list_products(products: pd.DataFrame, orders: pd.DataFrame) -> pd.DataFrame:
    out = (
        orders[orders["order_date"].dt.to_period("M") == '2020-02']
        .groupby("product_id", as_index=False)
        .agg({"unit": "sum"})
        .query("unit >= 100")
        .merge(
            products,
            on="product_id",
            how="inner"
        )
    )

    return out[["product_name", "unit"]]
```
