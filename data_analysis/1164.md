## Solution 1: joins
```
import pandas as pd

def price_at_given_date(products: pd.DataFrame) -> pd.DataFrame:
    product_ids = products[["product_id"]].drop_duplicates()
    products = products[products["change_date"] <= pd.Timestamp("2019-08-16")]
    max_dates = (
        products
        .groupby("product_id", as_index=False)
        .agg({"change_date": "max"})
    )

    out = (
        pd.merge(
            product_ids,
            max_dates,
            on=["product_id"],
            how="left",
        )
        .merge(
            products,
            on=["product_id" ,"change_date"],
            how="left"
        )
        .rename(columns={"new_price": "price"})
    )
    out["price"] = out["price"].fillna(10)
    
    return out[["product_id", "price"]]
```


## Solution 2: window functions:
```
def price_at_given_date(products: pd.DataFrame) -> pd.DataFrame:
    product_ids = products[["product_id"]].drop_duplicates()
    products = products[products["change_date"] <= pd.Timestamp("2019-08-16")]
    products.sort_values(by=["product_id", "change_date"], ascending=True, inplace=True)
    products["max_date"] = products.groupby("product_id")["change_date"].transform("last")
    products = products[products["change_date"] == products["max_date"]]
    out = (
        pd.merge(
            product_ids,
            products,
            on=["product_id"],
            how="left",
        )
        .rename(columns={"new_price": "price"})
    )
    out["price"] = out["price"].fillna(10)
    
    return out[["product_id", "price"]]
```
